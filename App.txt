  /**
   * Chama o servidor. Falhas de rede são repetidas automaticamente
   * (com o mesmo opId, então nada é gravado duas vezes).
   */
  function chamar(acao, dados, opt) {
    opt = opt || {};
    const tentativas = opt.tentativas || 3;
    if (typeof google === 'undefined' || !google.script) {
      return Promise.reject(Object.assign(new Error('Abra o sistema pelo link do App da Web do Google Apps Script.'), { codigo: 'SEM_GOOGLE' }));
    }
    return new Promise((resolve, reject) => {
      let n = 0;
      const tentar = () => {
        if (!navigator.onLine) { setSync('offline'); window.addEventListener('online', tentar, { once: true }); return; }
        n++; S.pendentes++; setSync('sync');
        google.script.run
          .withSuccessHandler(txt => {
            S.pendentes--;
            let r;
            try { r = JSON.parse(txt); } catch (e) { r = { ok: false, erro: { codigo: 'RESPOSTA', mensagem: 'Resposta inválida do servidor.' } }; }
            if (r.ok) {
              setSync(S.pendentes ? 'sync' : 'ok');
              if (r.versaoDados && !opt.semVersao) S.versaoDados = r.versaoDados;
              resolve(r.data);
            } else {
              if (r.erro.codigo === 'OCUPADO' && n < tentativas) { setTimeout(tentar, 900 * n); return; }
              setSync(S.pendentes ? 'sync' : 'ok');
              reject(Object.assign(new Error(r.erro.mensagem), { codigo: r.erro.codigo }));
            }
          })
          .withFailureHandler(() => {
            S.pendentes--;
            if (n < tentativas) { setSync('sync'); setTimeout(tentar, 1000 * Math.pow(2, n - 1)); return; }
            setSync(navigator.onLine ? 'erro' : 'offline');
            reject(Object.assign(new Error('Não foi possível falar com o servidor. Verifique a internet e tente de novo.'), { codigo: 'REDE' }));
          })
          .api(acao, dados || {});
      };
      tentar();
    });
  }

  // Atualização automática: verifica a cada 30 s se alguém da família alterou algo.

  function intervalo() {
    const p = S.periodo;
    if (p.modo === 'mes') return { inicio: isoDe(p.y, p.m, 1), fim: isoDe(p.y, p.m, ultimoDia(p.y, p.m)) };
    if (p.modo === 'ano') return { inicio: isoDe(p.y, 1, 1), fim: isoDe(p.y, 12, 31) };
    return { inicio: p.inicio, fim: p.fim };
  }
  function rotuloPeriodo() {
    const p = S.periodo;
    if (p.modo === 'mes') { const n = MESES[p.m - 1]; return n[0].toUpperCase() + n.slice(1) + ' de ' + p.y; }
    if (p.modo === 'ano') return String(p.y);
    return (p.inicio && p.fim) ? dataBR(p.inicio) + ' – ' + dataBR(p.fim) : 'Escolha as datas';
  }
  function moverPeriodo(n) {
    const p = S.periodo;
    if (p.modo === 'mes') { let t = p.y * 12 + (p.m - 1) + n; p.y = Math.floor(t / 12); p.m = t % 12 + 1; }
    else if (p.modo === 'ano') p.y += n;
    else if (p.inicio && p.fim) { const len = diasEntre(p.inicio, p.fim) + 1; p.inicio = addDias(p.inicio, n * len); p.fim = addDias(p.fim, n * len); }
  }

      if (b.dataset.modo !== 'custom' && p.modo === 'custom' && p.inicio) { p.y = +p.inicio.slice(0, 4); p.m = +p.inicio.slice(5, 7); }
      p.modo = b.dataset.modo; redesenhar(); recarregar();
    });
    const ini = $('#p-ini', el), fim = $('#p-fim', el);
    if (ini) [ini, fim].forEach(inp => inp.onchange = () => {
      S.periodo.inicio = ini.value; S.periodo.fim = fim.value;
      if (ini.value && fim.value && ini.value <= fim.value) { $('.rot', el).textContent = rotuloPeriodo(); recarregar(); }
    });
    const bf = $('#btn-filtros', el);
    if (bf) bf.onclick = () => { S.mostrarFiltros = !S.mostrarFiltros; redesenhar(); };
    const liga = (id, campo) => { const s = $('#' + id, el); if (s) s.onchange = () => { S.filtros[campo] = s.value; redesenhar(); recarregar(); }; };
    liga('f-pessoa', 'responsavel'); liga('f-cat', 'categoria'); liga('f-conta', 'conta_id');
    const lim = $('#f-limpar', el);
    if (lim) lim.onclick = () => { S.filtros = { responsavel: '', categoria: '', conta_id: '' }; redesenhar(); recarregar(); };
  }

  async function carregarDash() {
    const box = $('#dash'); if (!box) return;
    const i = intervalo();
    if (!i.inicio || !i.fim || i.inicio > i.fim) { box.innerHTML = '<div class="vazio"><div class="grande">📅</div><h3>Escolha o período</h3><p>Informe a data inicial e a final.</p></div>'; return; }
    const tk = ++S.tkDash;
    try {
      const d = await chamar('dashboard', { inicio: i.inicio, fim: i.fim, filtros: S.filtros });
      if (tk !== S.tkDash || S.rota !== 'inicio') return;
      S.dash = d;
      renderDash($('#dash'), d);
      atualizarSino();
    } catch (e) {
      if (tk !== S.tkDash) return;
      box.innerHTML = '<div class="vazio"><div class="grande">⚠️</div><h3>Não foi possível carregar</h3><p>' + esc(e.message) + '</p><button class="btn btn-pri" id="tentar">Tentar de novo</button></div>';
      $('#tentar').onclick = carregarDash;
    }
  }

  function statusPorData(v, temRestante, pagoTxt) {
    if (!temRestante) return ['paga', pagoTxt || 'Paga'];
    const h = hojeISO();
    if (v < h) return ['vencida', 'Vencida'];
    if (v === h) return ['hoje', 'Vence hoje'];
    if (v === addDias(h, 1)) return ['avencer', 'Vence amanhã'];
    if (v <= addDias(h, 7)) return ['avencer', 'A vencer'];
    return ['pendente', 'Pendente'];
  }
  const pagsDe = r => { if (Array.isArray(r.pagamentos)) return r.pagamentos; try { const a = JSON.parse(r.pagamentos || '[]'); return Array.isArray(a) ? a : []; } catch (e) { return []; } };
  const somaPags = r => pagsDe(r).reduce((s, p) => s + cents(p.valor), 0);
  /** Quanto falta pagar/receber (centavos). */
  function restanteDe(r, tipo) {
    if (r._restante != null) return r._restante;
    if (r.status === 'cancelada') return 0;
    const v = cents(r.valor);
    if (r.cartao_id) return r.status === 'paga' ? 0 : v;
    const p = pagsDe(r);
    if (p.length) return Math.max(0, v - somaPags(r));
    return (r.status === 'paga' || r.status === 'recebida') ? 0 : v;
  }
  function statusDespesa(r) {
    if (r.status === 'cancelada') return ['cancelada', 'Cancelada'];
    if (r.cartao_id) return r.status === 'paga' ? ['paga', 'Fatura paga'] : ['cartao', '💳 Fatura ' + mesAno(r.fatura)];
    const rest = restanteDe(r, 'despesas');
    const st = statusPorData(r.data_vencimento || r.data_compra, rest > 0);
    if (rest > 0 && pagsDe(r).length) return [st[0], st[1] + ' · parcial'];
    return st;
  }

  function ligarItens(el, lista, tipo, emModal) {
    $$('.item[data-id]', el).forEach(it => {
      const abrir = () => {
        const r = lista.find(x => x.id === it.dataset.id); if (!r) return;
        if (r._tipo === 'fatura') return abrirPagamento(r, 'faturas');
        if (r.cartao_id) return abrirCompraDeParcela(r);
        if (emModal === 'agenda' && restanteDe(r, tipo) > 0) return abrirPagamento(r, tipo);
        abrirForm(tipo === 'receitas' ? 'receita' : 'despesa', r);
      };
      it.onclick = e => { if (!e.target.closest('[data-rapido]')) abrir(); };
      it.onkeydown = e => { if (e.key === 'Enter') abrir(); };
    });
    $$('[data-rapido]', el).forEach(b => b.onclick = async e => {
      e.stopPropagation();
      const r = lista.find(x => x.id === b.dataset.rapido); if (!r) return;
      if (r._tipo === 'fatura') return abrirPagamento(r, 'faturas');
      b.disabled = true;
      try {
        await gravar('pagar', { entidade: tipo, id: r.id, valor: paraDecimal(restanteDe(r, tipo)), data: hojeISO(), conta_id: r.conta_id || '' });
        toast(tipo === 'receitas' ? 'Marcada como recebida' : 'Marcada como paga');
        if (emModal === true) fecharModal();
        recarregarVisao();
      } catch (err) { b.disabled = false; tratarErroGravacao(err); }
    });
  }

  function abrirPagamento(r, ent) {
    const rec = ent === 'receitas', fat = ent === 'faturas';
    const total = cents(r.valor), rest = restanteDe(r, ent);
    const pags = pagsDe(r);
    const verbo = rec ? 'Receber' : 'Pagar';
    const contaPadrao = r.conta_id || (fat ? (S.cartoes.find(k => k.id === r.cartao_id) || {}).conta_pagamento_id : '') || '';

'recebido' : 'pago') + ' (R$)</label><input type="text" inputmode="numeric" class="num valor-input" id="pg-valor" value="' + valorParaInput(rest) + '"><span class="ajuda">Para pagamento parcial, digite um valor menor.</span></div>' +

'Entrou na conta' : 'Saiu da conta') + '</label><select id="pg-conta"><option value="">Não informar</option>' + S.contas.filter(k => k.status !== 'inativa' || k.id === contaPadrao).map(k => '<option value="' + esc(k.id) + '"' + (k.id === contaPadrao ? ' selected' : '') + '>' + esc(k.nome) + '</option>').join('') + '</select></div>' : '') +
      (pags.length ? '<div class="rotulo" style="margin:6px 0">Pagamentos registrados</div><div class="lista">' + pags.map(p =>
        '<div class="item"><div class="item-ic">✓</div><div class="item-meio"><b class="num">' + brl(cents(p.valor)) + '</b><small>' + dataBR(p.data) + (p.conta_id ? ' · ' + esc(nomeConta(p.conta_id)) : '') + ' · por ' + esc(nomeMembro(p.por)) + '</small></div>' +
        (podeMexer && (ehAdmin() || p.por === S.usuario.email || !fat) ? '<div class="item-fim"><button class="item-acao" style="color:var(--vermelho)" data-rm="' + esc(p.id) + '">Desfazer</button></div>' : '') + '</div>').join('') + '</div>' : '') +
      (!rec && !fat && rest > 0 && podeMexer ? '<details class="mais" style="margin-top:12px"><summary>Adiar vencimento</summary><div style="display:flex;gap:8px"><input type="date" id="pg-adiar" value="' + esc(addDias(r.data_vencimento || r.data_compra, 7)) + '"><button class="btn btn-sec btn-peq" id="pg-adiar-btn">Adiar</button></div></details>' : '');
    const rodape = (fat ? '<button class="btn btn-sec btn-peq" id="pg-ver">Ver fatura</button>' : '<button class="btn btn-sec btn-peq" id="pg-editar">' + (podeMexer ? 'Editar lançamento' : 'Ver lançamento') + '</button>') +
      '<span class="espaco"></span>' + (rest > 0 && podeMexer ? '<button class="btn btn-pri" id="pg-ok">✓ ' + verbo + '</button>' : '');
    abrirModal({
      titulo: fat ? 'Pagar fatura' : rec ? 'Receber' : 'Pagar conta', corpo, rodape,
      aoAbrir: m => {
        const iv = $('#pg-valor', m); if (iv) mascaraValor(iv);
        const extra = fat ? { cartao_id: r.cartao_id, competencia: r.competencia } : { id: r.id };
        const ok = $('#pg-ok', m);
        if (ok) ok.onclick = async () => {
          const v = lerValor(iv);
          if (!v) return toast('Informe o valor.', 'erro');
          if (v > rest) return toast('O valor é maior que o que falta (' + brl(rest) + ').', 'erro');
          ok.disabled = true; ok.textContent = 'Salvando…';
          try {
            await gravar('pagar', Object.assign({ entidade: ent, valor: paraDecimal(v), data: $('#pg-data', m).value, conta_id: $('#pg-conta', m).value }, extra));
            fecharModal();
            toast(v < rest ? (rec ? 'Recebimento parcial registrado' : 'Pagamento parcial registrado') + ' · faltam ' + brl(rest - v) : (rec ? 'Recebido!' : 'Pago!'));
            recarregarVisao();
          } catch (e) { ok.disabled = false; ok.textContent = '✓ ' + verbo; tratarErroGravacao(e); }
        };
        $$('[data-rm]', m).forEach(b => b.onclick = async () => {
          if (!await confirmar('Desfazer este pagamento? O valor volta a ficar em aberto.', 'Desfazer', true)) return;
          try { await gravar('removerPagamento', Object.assign({ entidade: ent, pagamento_id: b.dataset.rm }, extra)); toast('Pagamento desfeito'); recarregarVisao(); }
          catch (e) { tratarErroGravacao(e); }
        });
        const ad = $('#pg-adiar-btn', m);
        if (ad) ad.onclick = async () => {
          const nova = $('#pg-adiar', m).value;
          try { await gravar('adiarVencimento', { id: r.id, data: nova }); fecharModal(); toast('Vencimento adiado para ' + dataBR(nova)); recarregarVisao(); }
          catch (e) { tratarErroGravacao(e); }
        };
        const ed = $('#pg-editar', m); if (ed) ed.onclick = () => abrirForm(rec ? 'receita' : 'despesa', r);
        const vf = $('#pg-ver', m); if (vf) vf.onclick = () => { fecharModal(); S.cartaoSel = r.cartao_id; S.compSel = r.competencia; if (location.hash === '#cartoes') rotear(); else location.hash = 'cartoes'; };
      }
    });
  }

  async function carregarLista(tipo) {
    const i = intervalo();
    if (!i.inicio || !i.fim) return;
    const tk = ++S.tkLista;
    try {
      const l = await chamar('listar', { entidade: tipo, inicio: i.inicio, fim: i.fim });
      if (tk !== S.tkLista || S.rota !== tipo) return;
      S.lista = l;
      renderLista(tipo);
    } catch (e) {
      const box = $('#lista'); if (box) box.innerHTML = vazioHTML('⚠️', 'Não foi possível carregar', e.message);
    }
  }

  function renderLista(tipo) {
    const box = $('#lista'); if (!box) return;
    const rec = tipo === 'receitas';
    const todos = S.lista.filter(r => r.status !== 'cancelada');
    const total = todos.reduce((s, r) => s + cents(r.valor), 0);
    const feito = todos.filter(r => r.status === (rec ? 'recebida' : 'paga')).reduce((s, r) => s + cents(r.valor), 0);
    $('#resumo').innerHTML =
      '<div class="painel"><small>Total</small><b class="num" style="color:var(--' + (rec ? 'verde' : 'vermelho') + ')">' + brl(total) + '</b></div>' +
      '<div class="painel"><small>' + (rec ? 'Recebido' : 'Pago') + '</small><b class="num">' + brl(feito) + '</b></div>' +
      '<div class="painel"><small>' + (rec ? 'A receber' : 'Em aberto') + '</small><b class="num">' + brl(total - feito) + '</b></div>';
    let l = S.lista;
    const f = S.filtroStatus;
    if (f !== 'todos') l = l.filter(r => rec ? r.status === f : (f === 'vencida' ? statusDespesa(r)[0] === 'vencida' : r.status === f));
    if (!l.length) {
      box.innerHTML = '<div class="lista">' + vazioHTML(rec ? '💰' : '💸', S.lista.length ? 'Nenhum lançamento com este filtro' : (rec ? 'Nenhuma receita neste período' : 'Nenhuma despesa neste período'),
        rec ? 'Registre salários, vendas e outras entradas para acompanhar quanto a família recebe.' : 'Registre as compras e contas para saber para onde o dinheiro está indo.',
        podeCriar() && !S.lista.length ? '<button class="btn btn-pri" id="vazio-novo">+ ' + (rec ? 'Registrar receita' : 'Registrar despesa') + '</button>' : '') + '</div>';
      const b = $('#vazio-novo'); if (b) b.onclick = () => abrirForm(rec ? 'receita' : 'despesa');
      return;
    }
    const campo = r => rec ? r.data : r.data_compra;
    let html = '', dia = '';
    l.forEach(r => {
      const d = campo(r);
      if (d !== dia) { dia = d; const dt = new Date(d + 'T12:00:00'); html += '<div class="lista-dia">' + dt.toLocaleDateString('pt-BR', { weekday: 'long', day: '2-digit', month: 'long' }) + '</div>'; }
      html += itemHTML(r, tipo);
    });
    box.innerHTML = '<div class="lista">' + html + '</div>';
    ligarItens(box, l, tipo, false);
  }

  let tNotif = null, seqNotif = 0, seqAplicado = 0;
  /** Chama o servidor e só aplica a resposta se nenhuma mais nova já tiver chegado. */
  async function pedirNotif(acao, dados) {
    const meu = ++seqNotif;
    const r = await chamar(acao, dados || {}, { semVersao: true, tentativas: acao === 'notificacoes' ? 1 : 3 });
    if (meu >= seqAplicado) { seqAplicado = meu; S.notif = r; atualizarSino(); if (S.modalNotif) renderCentral(); }
    return r;
  }
  function carregarNotif(imediato) {
    clearTimeout(tNotif);
    tNotif = setTimeout(() => { pedirNotif('notificacoes').catch(() => { /* o indicador de sincronização já mostra problemas */ }); }, imediato ? 0 : 600);
  }

  function atualizarSino() {
    const b = $('#btn-sino'); if (!b) return;
    const n = S.notif ? S.notif.naoLidas : 0;
    const bol = $('.bolinha', b);
    if (n && !bol) b.insertAdjacentHTML('beforeend', '<span class="bolinha">' + (n > 99 ? '99+' : n) + '</span>');
    else if (bol) { if (n) bol.textContent = n > 99 ? '99+' : n; else bol.remove(); }
    b.title = n ? plural(n, 'aviso não lido', 'avisos não lidos') : 'Avisos';
  }

  async function irNotificacao(x) {
    // Marca como lida ao abrir.
    if (!x.lida) pedirNotif('marcarNotificacoes', { chaves: [x.chave] }).catch(() => {});
    const a = x.acao || {};
    if (a.duplicado) return abrirDuplicado(x);
    S.modalNotif = null; fecharModal();
    if (a.rota === 'cartoes') { S.cartaoSel = a.cartao_id || S.cartaoSel; S.compSel = a.comp || ''; }
    if (a.rota === 'pagar' && a.aba) S.abaAgenda = a.aba;
    const hash = a.rota === 'metas' ? 'metas?aba=' + (a.aba || 'metas') : a.rota;
    if (a.rota === 'metas') S.abaPlan = a.aba || 'metas';
    if (('#' + hash) === location.hash) rotear(); else location.hash = hash;
  }

  function resumoHistorico(h) {
    const d = h.detalhes || {};
    if (h.acao === 'editar' || h.acao === 'alterar_membro') {
      const ks = Object.keys(d).filter(k => Array.isArray(d[k]));
      return ks.slice(0, 4).map(k => (CAMPOS_TXT[k] || k) + ': ' + fmtValHist(k, d[k][0]) + ' → ' + fmtValHist(k, d[k][1])).join(' · ') + (ks.length > 4 ? ' · +' + (ks.length - 4) : '');
    }
    if (h.acao === 'edicao_manual') return 'Aba ' + (d.aba || '') + ', célula ' + (d.celula || '') + (d.linhas > 1 ? ' (' + d.linhas + ' linhas)' : '');
    if (h.acao === 'automacao') return d.erro ? 'Erro: ' + d.erro : plural(d.recorrencias || 0, 'lançamento criado', 'lançamentos criados') + (d.backup ? ' · cópia feita' : '');
    const partes = [];
    if (d.descricao || d.nome) partes.push(d.descricao || d.nome);
    if (d.valor) partes.push(brl(cents(d.valor)));
    if (d.email) partes.push(d.email);
    if (d.cartao) partes.push(d.cartao);
    if (d.data && h.acao !== 'criar') partes.push(dataBR(d.data));
    if (d.parcelas) partes.push(d.parcelas + 'x');
    if (d.de && d.para) partes.push(mesAno(d.de) + ' → ' + mesAno(d.para));
    return partes.join(' · ');
  }
  function fmtValHist(k, v) {
    if (v === '' || v == null) return '—';
    if (/valor|limite|saldo/.test(k) && /^-?\d+(\.\d+)?$/.test(String(v))) return brl(cents(v));
    if (/^\d{4}-\d{2}-\d{2}$/.test(String(v))) return dataBR(v);
    if (k === 'conta_id') return nomeConta(v);
    return String(v).slice(0, 40);
  }

  const addMesYM = (y, m, n) => { const t = y * 12 + (m - 1) + n; return { y: Math.floor(t / 12), m: t % 12 + 1 }; };
  const addComp = (comp, n) => { const p = addMesYM(+comp.slice(0, 4), +comp.slice(5, 7), n); return p.y + '-' + pad(p.m); };
  const mesAno = comp => comp ? MES_CURTO[+comp.slice(5, 7) - 1] + '/' + comp.slice(2, 4) : '';
  const mesAnoLongo = comp => { const n = MESES[+comp.slice(5, 7) - 1]; return n[0].toUpperCase() + n.slice(1) + ' de ' + comp.slice(0, 4); };
  function competenciaCompra(k, iso) {
    const F = +k.dia_fechamento || 1, V = +k.dia_vencimento || 10;
    const y = +iso.slice(0, 4), m = +iso.slice(5, 7), d = +iso.slice(8, 10);
    const cm = d < Math.min(F, ultimoDia(y, m)) ? { y, m } : addMesYM(y, m, 1);
    const vm = V > F ? cm : addMesYM(cm.y, cm.m, 1);
    return vm.y + '-' + pad(vm.m);
  }
  function datasFatura(k, comp) {
    const F = +k.dia_fechamento || 1, V = +k.dia_vencimento || 10;
    const y = +comp.slice(0, 4), m = +comp.slice(5, 7);
    const cm = V > F ? { y, m } : addMesYM(y, m, -1);
    return { fechamento: isoDe(cm.y, cm.m, Math.min(F, ultimoDia(cm.y, cm.m))), vencimento: isoDe(y, m, Math.min(V, ultimoDia(y, m))) };
  }

  async function carregarAgenda() {
    const i = intervalo(); if (!i.inicio || !i.fim) return;
    const tk = ++S.tkLista;
    try {
      const r = await chamar('agenda', { inicio: i.inicio, fim: i.fim });

'Pagas' : 'Recebidas'], ['todos', 'Todas']].map(x => '<button data-st="' + x[0] + '" class="' + (S.statusAgenda === x[0] ? 'ativo' : '') + '">' + x[1] + '</button>').join('') + '</div>' +
      sel('ag-pessoa', 'Pessoa', pessoas, fil.pessoa) + sel('ag-cat', 'Categoria', cats, fil.categoria) +
      '<div class="segmentado"><button data-vis="lista" class="' + (S.visaoAgenda === 'lista' ? 'ativo' : '') + '">☰ Lista</button><button data-vis="cal" class="' + (S.visaoAgenda === 'cal' ? 'ativo' : '') + '">📅 Calendário</button></div>' +
      (podeCriar() ? '<span style="flex:1"></span><button class="btn btn-pri btn-peq" id="ag-novo">+ ' + (pagar ? 'Conta a pagar' : 'Conta a receber') + '</button>' : '');
    $$('[data-st]', $('#ag-filtros')).forEach(b => b.onclick = () => { S.statusAgenda = b.dataset.st; renderAgenda(); });

    $('#ag-cat').onchange = e => { fil.categoria = e.target.value; renderAgenda(); };

    const st = S.statusAgenda;
    let l = st === 'aberto' ? aberto : st === 'atrasado' ? atrasado : st === 'quitado' ? base.filter(x => x._restante <= 0) : base;

    if (S.visaoAgenda === 'cal') return renderCalendario(box, base, tipo);

'Contas com vencimento, boletos e faturas de cartão aparecem aqui. Use “+ Conta a pagar” para registrar uma.' : 'Receitas previstas aparecem aqui até serem recebidas.') + '</div>';
      return;
    }
    // Grupos por urgência
    const amanha = addDias(hoje, 1), d3 = addDias(hoje, 3), d7 = addDias(hoje, 7);
    const grupos = [['Atrasadas', x => x._restante > 0 && x._venc < hoje], ['Vencem hoje', x => x._restante > 0 && x._venc === hoje], ['Vencem amanhã', x => x._restante > 0 && x._venc === amanha],

    const usado = new Set();
    let html = '';
    grupos.forEach(g => {
      const itens = l.filter(x => !usado.has(x) && g[1](x));
      if (!itens.length) return;
      itens.forEach(x => usado.add(x));
      html += '<div class="lista-dia">' + g[0] + ' · ' + brl(itens.reduce((s, x) => s + (x._restante > 0 ? x._restante : cents(x.valor)), 0)) + '</div>' + itens.map(x => itemAgenda(x, tipo)).join('');
    });
    box.innerHTML = '<div class="lista">' + html + '</div>';
    ligarItens(box, l, tipo, 'agenda');
  }

  function itemAgenda(x, tipo) {
    if (x._tipo === 'fatura') return itemHTML(x, tipo);
    const h = itemHTML(x, tipo);
    if (x._restante <= 0) return h;
    const dias = diasEntre(hojeISO(), x._venc);
    const txt = dias < 0 ? plural(-dias, 'dia de atraso', 'dias de atraso') : dias === 0 ? 'vence hoje' : 'faltam ' + plural(dias, 'dia', 'dias');
    return h.replace('</small></div><div class="item-fim">', ' · <b style="color:var(--' + (dias < 0 ? 'vermelho' : dias <= 3 ? 'laranja' : 'txt-2') + ')">' + txt + '</b></small></div><div class="item-fim">');
  }

  function renderCalendario(box, itens, tipo) {
    const p = S.periodo;
    const base = p.modo === 'mes' ? { y: p.y, m: p.m } : { y: +intervalo().inicio.slice(0, 4), m: +intervalo().inicio.slice(5, 7) };
    const primeiro = new Date(base.y, base.m - 1, 1).getDay();
    const nd = ultimoDia(base.y, base.m), hoje = hojeISO();
    const porDia = {};
    itens.forEach(x => { const d = x._restante > 0 ? x._venc : (x.data_pagamento || x._venc); (porDia[d] = porDia[d] || []).push(x); });
    let cel = '';
    for (let i = 0; i < primeiro; i++) cel += '<div class="cal-dia vazio"></div>';
    for (let d = 1; d <= nd; d++) {
      const iso = isoDe(base.y, base.m, d), xs = porDia[iso] || [];
      const aberto = xs.filter(x => x._restante > 0).reduce((s, x) => s + x._restante, 0);
      const pago = xs.filter(x => x._restante <= 0).length;
      const atras = xs.some(x => x._restante > 0 && x._venc < hoje);
      cel += '<button class="cal-dia' + (iso === hoje ? ' hoje' : '') + (xs.length ? ' tem' : '') + '" data-dia="' + iso + '"><span class="n">' + d + '</span>' +
        (aberto ? '<span class="v num ' + (atras ? 'atras' : S.abaAgenda === 'pagar' ? 'pagar' : 'receber') + '">' + curtoBR(aberto) + '</span>' : '') +
        (pago ? '<span class="ok">✓' + (pago > 1 ? pago : '') + '</span>' : '') + '</button>';
    }
    box.innerHTML = '<div class="painel"><h3 style="margin-bottom:10px">' + esc(MESES[base.m - 1][0].toUpperCase() + MESES[base.m - 1].slice(1)) + ' de ' + base.y + '</h3>' +
      '<div class="cal">' + ['Dom', 'Seg', 'Ter', 'Qua', 'Qui', 'Sex', 'Sáb'].map(x => '<div class="cal-sem">' + x + '</div>').join('') + cel + '</div>' +
      '<p class="mudo pequeno" style="margin:10px 0 0">Valores em aberto por dia · <span style="color:var(--vermelho)">vermelho</span> = atrasado · ✓ = quitado. Toque num dia para ver os lançamentos.</p></div>';
    $$('[data-dia]', box).forEach(b => b.onclick = () => {
      const iso = b.dataset.dia, xs = porDia[iso] || [];
      abrirModal({
        titulo: new Date(iso + 'T12:00:00').toLocaleDateString('pt-BR', { weekday: 'long', day: '2-digit', month: 'long' }), largo: true,
        corpo: (xs.length ? '<div class="lista">' + xs.map(x => itemAgenda(x, tipo)).join('') + '</div>' : vazioHTML('📅', 'Nada neste dia', '')) ,
        rodape: podeCriar() ? '<span class="espaco"></span><button class="btn btn-sec btn-peq" id="cd-rec">+ Receita neste dia</button><button class="btn btn-pri btn-peq" id="cd-des">+ Conta neste dia</button>' : '',
        aoAbrir: m => {
          ligarItens(m, xs, tipo, 'agenda');
          const a = $('#cd-des', m); if (a) a.onclick = () => abrirForm('despesa', null, { status: 'pendente', data_compra: iso <= hojeISO() ? iso : hojeISO(), data_vencimento: iso });
          const r = $('#cd-rec', m); if (r) r.onclick = () => abrirForm('receita', null, { status: iso <= hojeISO() ? 'recebida' : 'prevista', data: iso });
        }
      });
    });
  }

  /** Nova compra no cartão (à vista ou parcelada) ou edição de uma compra existente. */
  function abrirCompra(pre, existente) {
    if (!podeCriar() && !existente) return;
    const ed = !!existente;
    const c = Object.assign({ data_compra: hojeISO(), parcelas: 1, responsavel: S.usuario.nome }, pre || {}, existente || {});
    const ativos = S.cartoes.filter(k => k.status !== 'inativo' || k.id === c.cartao_id);
    if (!c.cartao_id && ativos.length) c.cartao_id = ativos[0].id;

  async function carregarPlanejamento() {
    const box = $('#plan-box'); if (!box) return;
    const tk = ++S.tkLista;
    try {
      if (S.abaPlan === 'orcamento') {
        const o = await chamar('orcamento', { competencia: S.compOrc });
        if (tk !== S.tkLista || S.rota !== 'metas') return;
        S.orc = o; renderOrcamento();
      } else {
        const p = await chamar('planejamento', {});
        if (tk !== S.tkLista || S.rota !== 'metas') return;
        S.plan = p;
        if (S.abaPlan === 'dividas') renderDividas(); else renderMetas();

    } catch (e) { box.innerHTML = vazioHTML('⚠️', 'Não foi possível carregar', e.message); }
  }

  function periodoComparacao() {
    const i = intervalo(), modo = S.relComparar;
    if (!modo) return null;
    if (modo === 'custom') return S.relB.inicio && S.relB.fim ? { inicio: S.relB.inicio, fim: S.relB.fim } : null;
    const y0 = +i.inicio.slice(0, 4), m0 = +i.inicio.slice(5, 7), y1 = +i.fim.slice(0, 4), m1 = +i.fim.slice(5, 7);
    const mesesInteiros = i.inicio.slice(8) === '01' && i.fim === isoDe(y1, m1, ultimoDia(y1, m1));
    if (modo === 'ano') {
      const umAnoAntes = iso => { const y = +iso.slice(0, 4) - 1, m = +iso.slice(5, 7); return isoDe(y, m, Math.min(+iso.slice(8), ultimoDia(y, m))); };
      return { inicio: umAnoAntes(i.inicio), fim: mesesInteiros ? isoDe(y1 - 1, m1, ultimoDia(y1 - 1, m1)) : umAnoAntes(i.fim) };
    }
    if (mesesInteiros) {
      const n = (y1 * 12 + m1) - (y0 * 12 + m0) + 1;
      const a = addMesYM(y0, m0, -n), b = addMesYM(y1, m1, -n);
      return { inicio: isoDe(a.y, a.m, 1), fim: isoDe(b.y, b.m, ultimoDia(b.y, b.m)) };
    }
    const len = diasEntre(i.inicio, i.fim) + 1;
    return { inicio: addDias(i.inicio, -len), fim: addDias(i.inicio, -1) };
  }

  async function carregarRelatorio() {
    const box = $('#rel-box'); if (!box) return;
    const i = intervalo(); if (!i.inicio || !i.fim || i.inicio > i.fim) return;
    const tk = ++S.tkLista;
    try {
      const r = await chamar('relatorio', paramsRelatorio());
      if (tk !== S.tkLista || S.rota !== 'relatorios') return;
      S.rel = r; renderRelatorio();
    } catch (e) { box.innerHTML = vazioHTML('⚠️', 'Não foi possível gerar o relatório', e.message); }
  }

    if (tp === 'data') return dataBR(v);

      const r = await chamar('exportarRelatorio', paramsRelatorio());
      abrirModal({
        titulo: 'Relatório pronto',
        corpo: '<p>Criamos a planilha <b>' + esc(r.nome) + '</b> no seu Google Drive. Escolha como baixar:</p>' +
          '<div class="menu-rapido">' +
          '<a class="btn ' + (fmt === 'xlsx' ? 'btn-pri' : 'btn-borda') + '" href="' + esc(r.xlsx) + '" target="_blank" rel="noopener">📊 Baixar Excel (.xlsx)</a>' +
          '<a class="btn ' + (fmt === 'pdf' ? 'btn-pri' : 'btn-borda') + '" href="' + esc(r.pdf) + '" target="_blank" rel="noopener">📄 Baixar PDF</a>' +
          '<a class="btn ' + (fmt === 'csv' ? 'btn-pri' : 'btn-borda') + '" href="' + esc(r.csv) + '" target="_blank" rel="noopener">📑 Baixar CSV</a>' +
          '<a class="btn btn-sec" href="' + esc(r.url) + '" target="_blank" rel="noopener">Abrir no Google Planilhas</a></div>' +
          '<p class="mudo pequeno">O download abre numa nova aba. A planilha fica no seu Drive — apague quando não precisar mais.</p>'
      });
    } catch (e) { erroToast(e); }
    botao.disabled = false; botao.textContent = txt;
  }

  // ---------------------------------------------------------
  // Telas de entrada: configurar família, convite, sem acesso
  // ---------------------------------------------------------
  function aplicarPacote(p) {
    S.usuario = p.usuario; S.familia = p.familia; S.perfil = p.usuario.perfil;
    S.membros = p.membros; S.categorias = p.categorias; S.contas = p.contas; S.sugestoes = p.sugestoes || []; S.cartoes = p.cartoes || [];
    S.urlApp = p.urlApp; S.urlPlanilha = p.urlPlanilha; S.versaoApp = p.versaoApp;
  }

  function tratarEstado(r) {
    if (r.estado !== 'pronto') { clearInterval(S.timer); S.usuario = null; fecharModal(); }
    if (r.estado === 'configurar') return telaCriarFamilia(r);
    if (r.estado === 'convite') return telaConvite(r);
    if (r.estado === 'negado') return telaNegado(r);
    const primeira = !S.usuario;
    aplicarPacote(r);
    renderShell();
    rotear();
    if (primeira) { iniciarVerificacao(); carregarNotif(true); }
  }

  function telaCriarFamilia(r) {
    if (!r.podeCriar) {
      $('#app').innerHTML = '<div class="entrada"><div class="entrada-card"><div class="logo">R$</div><h1>Quase lá</h1><p>O administrador ainda não criou a família neste sistema. Peça para ele abrir o link primeiro e depois enviar um convite para <b>' + esc(r.email) + '</b>.</p><button class="btn btn-sec btn-bloco" id="denovo">Tentar de novo</button></div></div>';
      $('#denovo').onclick = iniciar;
      return;
    }
    $('#app').innerHTML = '<div class="entrada"><div class="entrada-card"><div class="logo">R$</div><h1>Vamos organizar as finanças da família</h1><p>Leva menos de um minuto. Depois você convida as outras pessoas.</p>' +
      '<div class="campo"><label for="s-fam">Nome da família</label><input type="text" id="s-fam" maxlength="80" placeholder="Ex.: Família Silva"></div>' +
      '<div class="campo"><label for="s-nome">Seu nome</label><input type="text" id="s-nome" maxlength="60" placeholder="Como você quer aparecer"></div>' +
      '<fieldset><legend>🏦 Primeira conta (opcional)</legend>' +
      '<div class="campo"><label for="s-conta">Nome da conta</label><input type="text" id="s-conta" maxlength="60" placeholder="Ex.: Conta do Nubank"></div>' +
      '<div class="linha-2"><div class="campo"><label for="s-saldo">Saldo atual (R$)</label><input type="text" inputmode="numeric" class="num" id="s-saldo" placeholder="0,00"></div>' +
      '<div class="campo"><label for="s-tipo">Tipo</label><select id="s-tipo">' + TIPOS_CONTA.map(t => '<option value="' + t[0] + '">' + t[1] + '</option>').join('') + '</select></div></div></fieldset>' +
      '<label class="check" style="margin-bottom:18px"><input type="checkbox" id="s-demo"><span>Incluir dados de demonstração <small class="mudo">(fictícios, marcados com [DEMO] e removíveis a qualquer momento)</small></span></label>' +
      '<button class="btn btn-pri btn-bloco" id="s-criar">Criar família</button>' +
      '<p class="mudo pequeno" style="text-align:center;margin-top:14px">Conectado como ' + esc(r.email) + '</p></div></div>';
    mascaraValor($('#s-saldo'));
    $('#s-criar').onclick = async e => {
      const b = e.currentTarget;
      const fam = $('#s-fam').value.trim(), nome = $('#s-nome').value.trim();
      if (!fam || !nome) return toast('Preencha o nome da família e o seu nome.', 'erro');
      b.disabled = true; b.textContent = 'Criando…';
      const conta = $('#s-conta').value.trim() ? { nome: $('#s-conta').value.trim(), tipo: $('#s-tipo').value, saldo_inicial: paraDecimal(lerValor($('#s-saldo'))), data_saldo_inicial: hojeISO(), titular: nome } : null;
      try {
        const p = await gravar('criarFamilia', { nomeFamilia: fam, nomeUsuario: nome, conta, demo: $('#s-demo').checked });
        toast('Família criada! Bem-vindo(a) 🎉');
        tratarEstado(p);
      } catch (err) { erroToast(err); b.disabled = false; b.textContent = 'Criar família'; }
    };
  }

  function telaConvite(r) {
    $('#app').innerHTML = '<div class="entrada"><div class="entrada-card"><div class="logo">R$</div><h1>Você foi convidado(a)!</h1>' +
      '<p><b>' + esc(r.convite.convidado_por) + '</b> convidou você para as finanças da <b>' + esc(r.familia.nome) + '</b> como <b>' + rotuloPerfil(r.convite.perfil) + '</b>.</p>' +
      '<div class="campo"><label for="cv-nome">Como você quer aparecer</label><input type="text" id="cv-nome" maxlength="60" value="' + esc(r.convite.nome || '') + '"></div>' +
      '<button class="btn btn-pri btn-bloco" id="cv-aceitar">Aceitar convite</button><p class="mudo pequeno" style="text-align:center;margin-top:14px">Conectado como ' + esc(r.email) + '</p></div></div>';
    $('#cv-aceitar').onclick = async e => {
      const b = e.currentTarget; b.disabled = true;
      try { const p = await gravar('aceitarConvite', { nome: $('#cv-nome').value }); toast('Bem-vindo(a) à família!'); tratarEstado(p); }
      catch (err) { erroToast(err); b.disabled = false; }
    };
  }

  function telaNegado(r) {
    $('#app').innerHTML = '<div class="entrada"><div class="entrada-card"><div class="logo" style="background:var(--laranja)">🔒</div><h1>Acesso não liberado</h1><p>' + esc(r.motivo) + '</p>' +
      '<div class="formula"><div class="l"><span>Conta conectada</span><b>' + esc(r.email) + '</b></div></div>' +
      '<p class="mudo pequeno">Se o convite foi enviado para outro e-mail, troque de conta Google no navegador e abra o link novamente.</p>' +
      '<button class="btn btn-sec btn-bloco" id="denovo">Tentar de novo</button></div></div>';
    $('#denovo').onclick = iniciar;
  }

  function telaErroFatal(e) {
    $('#app').innerHTML = '<div class="entrada"><div class="entrada-card"><div class="logo" style="background:var(--vermelho)">!</div><h1>Não foi possível abrir</h1><p>' + esc(e.message) + '</p><button class="btn btn-pri btn-bloco" id="denovo">Tentar de novo</button></div></div>';
    $('#denovo').onclick = iniciar;
  }

  async function iniciar() {
    $('#app').innerHTML = '<div class="splash"><div class="splash-logo">R$</div><div class="splash-txt">Carregando suas finanças…</div></div>';
    try { tratarEstado(await chamar('iniciar')); }
    catch (e) { telaErroFatal(e); }
  }

  iniciar();
})();
</script>
